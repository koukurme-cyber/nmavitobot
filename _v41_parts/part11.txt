    if not text.startswith("/"):
        return

    command = text.split(maxsplit=1)[0].split("@", 1)[0].casefold()

    # Команда пользователя исчезает по той же настройке, что и ответы бота.
    delete_message_later(chat_id, message["message_id"])

    if command == "/delete_after":
        parts = text.split(maxsplit=1)

        if len(parts) == 1:
            _send_delete_after_settings(chat_id)
            return

        raw = parts[1].strip()
        try:
            minutes = int(raw)
        except ValueError:
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": "Укажите число минут, например <code>/delete_after 45</code>.",
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        if minutes < 0 or minutes > 10080:
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": "Допустимо от 0 до 10080 минут.",
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        all_settings, settings = _get_chat_notify_settings(chat_id)
        settings["delete_after_minutes"] = minutes
        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)
        _send_delete_after_settings(chat_id)
        return

    if command == "/start":
        _send_help(chat_id)
        return

    mode, seller_key = _parse_direct_command(command)
    if mode:
        send_query(chat_id, mode, seller_key)
        return

    if command in {"/help", "/commands"}:
        _send_help(chat_id)
