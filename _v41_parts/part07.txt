        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)
        return False

    if text.casefold() in {"отмена", "cancel", "/cancel"}:
        _clear_pending_time_input(settings)
        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)

        try:
            if prompt_message_id:
                telegram_api(
                    "deleteMessage",
                    {"chat_id": chat_id, "message_id": prompt_message_id},
                )
        except Exception:
            pass

        delete_message_later(chat_id, message["message_id"], delay=1)

        if menu_message_id:
            _edit_notify_message(
                chat_id,
                menu_message_id,
                _collector_text(settings, collector),
                _collector_keyboard(settings, collector),
            )
        return True

    value = _parse_direct_time(text)
    if not value:
        error = telegram_api(
            "sendMessage",
            {
                "chat_id": chat_id,
                "text": (
                    "Неверный формат. Введите время как <b>09:15</b> "
                    "или отправьте <b>отмена</b>."
                ),
                "parse_mode": "HTML",
                "reply_markup": json.dumps(
                    {
                        "force_reply": True,
                        "selective": True,
                        "input_field_placeholder": "ЧЧ:ММ",
                    },
                    ensure_ascii=False,
                ),
            },
        )
        pending["prompt_message_id"] = error["message_id"]
        settings["pending_time_input"] = pending
        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)

        delete_message_later(chat_id, message["message_id"], delay=1)
        delete_message_later(chat_id, error["message_id"])
        return True

    cfg = _collector_cfg(settings, collector)
    times = list(cfg.get("times") or [])

    if edit_index is not None:
        try:
            edit_index = int(edit_index)
        except Exception:
            edit_index = None

    if edit_index is not None and 0 <= edit_index < len(times):
        times[edit_index] = value
    elif value not in times:
        if len(times) >= 8:
            warning = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": "Для этого сборщика уже сохранено 8 времён.",
                },
            )
            delete_message_later(chat_id, warning["message_id"])
            delete_message_later(chat_id, message["message_id"], delay=1)
            return True
        times.append(value)

    cfg["times"] = sorted(set(times))
    _clear_pending_time_input(settings)

    all_settings[str(chat_id)] = settings
    _save_notify_settings(all_settings)

    try:
        if prompt_message_id:
            telegram_api(
                "deleteMessage",
                {"chat_id": chat_id, "message_id": prompt_message_id},
            )
    except Exception:
        pass

    # Введённое пользователем время убираем почти сразу.
    delete_message_later(chat_id, message["message_id"], delay=1)

    if menu_message_id:
        _edit_notify_message(
            chat_id,
            menu_message_id,
            _collector_text(settings, collector),
            _collector_keyboard(settings, collector),
        )

    confirmation = telegram_api(
        "sendMessage",
        {
            "chat_id": chat_id,
            "text": f"Время сохранено: <b>{html.escape(value)}</b>",
            "parse_mode": "HTML",
        },
    )
    delete_message_later(chat_id, confirmation["message_id"], delay=5)
    return True


def _send_notify_settings(chat_id):
    all_settings, settings = _get_chat_notify_settings(chat_id)
    message = telegram_api(
        "sendMessage",
        {
            "chat_id": chat_id,
            "text": _notify_overview_text(settings),
            "parse_mode": "HTML",
            "reply_markup": json.dumps(
                _notify_overview_keyboard(settings),
                ensure_ascii=False,
            ),
        },
    )
    delete_message_later(chat_id, message["message_id"])
    return message


def _edit_notify_message(chat_id, message_id, text, keyboard):
    try:
        telegram_api(
            "editMessageText",
            {
                "chat_id": chat_id,
                "message_id": message_id,
                "text": text,
                "parse_mode": "HTML",
                "reply_markup": json.dumps(keyboard, ensure_ascii=False),
            },
        )
    except Exception as exc:
        if "message is not modified" not in str(exc).lower():
            print("NOTIFY EDIT ERROR:", exc, flush=True)
