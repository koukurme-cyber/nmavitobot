DAY_LABELS = ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"]

NOTIFY_COLLECTORS = {
    "full": {
        "title": "Вся информация",
        "short": "Статус",
        "emoji": "📋",
    },
    "finance": {
        "title": "Баланс",
        "short": "Баланс",
        "emoji": "💰",
    },
    "ads": {
        "title": "Объявления",
        "short": "Объявления",
        "emoji": "📣",
    },
}


def _default_collector_settings():
    return {
        "enabled": False,
        "days": [0, 1, 2, 3, 4, 5, 6],
        "times": [],
        "last_sent": {},
    }


def _default_notify_settings():
    return {
        "collectors": {
            key: _default_collector_settings()
            for key in NOTIFY_COLLECTORS
        },
        "delete_after_minutes": DEFAULT_DELETE_AFTER_MINUTES,
        "alarm_editor": {},
    }


def _load_notify_settings():
    try:
        if not NOTIFY_SETTINGS_PATH.exists():
            return {}
        with NOTIFY_SETTINGS_PATH.open("r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, dict) else {}
    except Exception as exc:
        print("NOTIFY SETTINGS READ ERROR:", exc, flush=True)
        return {}


def _save_notify_settings(settings):
    tmp = NOTIFY_SETTINGS_PATH.with_suffix(".tmp")
    try:
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(settings, f, ensure_ascii=False, indent=2)
        tmp.replace(NOTIFY_SETTINGS_PATH)
    except Exception as exc:
        print("NOTIFY SETTINGS WRITE ERROR:", exc, flush=True)


def _migrate_chat_notify_settings(current):
    """
    v35-v37 stored one global notification schedule per chat.
    v38 moves it into collectors.full and keeps finance/ads disabled.
    """
    if not isinstance(current, dict):
        return _default_notify_settings(), True

    changed = False

    if "collectors" not in current:
        old_enabled = bool(current.get("enabled", False))
        old_days = current.get("days", [0, 1, 2, 3, 4, 5, 6])
        old_times = current.get("times", [])
        old_last_sent = current.get("last_sent", {})

        current["collectors"] = {
            "full": {
                "enabled": old_enabled,
                "days": old_days if isinstance(old_days, list) else [0, 1, 2, 3, 4, 5, 6],
                "times": old_times if isinstance(old_times, list) else [],
                "last_sent": old_last_sent if isinstance(old_last_sent, dict) else {},
            },
            "finance": _default_collector_settings(),
            "ads": _default_collector_settings(),
        }

        for legacy_key in ("enabled", "days", "times", "last_sent"):
            current.pop(legacy_key, None)
        changed = True

    collectors = current.setdefault("collectors", {})
    for key in NOTIFY_COLLECTORS:
        cfg = collectors.get(key)
        if not isinstance(cfg, dict):
            collectors[key] = _default_collector_settings()
            changed = True
            continue

        defaults = _default_collector_settings()
        for field, default_value in defaults.items():
            if field not in cfg:
                cfg[field] = default_value.copy() if isinstance(default_value, (list, dict)) else default_value
                changed = True

    if "delete_after_minutes" not in current:
        current["delete_after_minutes"] = DEFAULT_DELETE_AFTER_MINUTES
        changed = True

    if not isinstance(current.get("alarm_editor"), dict):
        current["alarm_editor"] = {}
        changed = True

    return current, changed


def _get_chat_notify_settings(chat_id):
    all_settings = _load_notify_settings()
    key = str(chat_id)
    current = all_settings.get(key)

    current, changed = _migrate_chat_notify_settings(current)
    all_settings[key] = current

    if changed:
        _save_notify_settings(all_settings)

    return all_settings, current


def _normalize_times(values):
    result = []
    for value in values:
        value = str(value).strip()
        try:
            parsed = datetime.strptime(value, "%H:%M")
        except ValueError:
            raise ValueError(f"Неверное время: {value}. Нужен формат ЧЧ:ММ.")
        normalized = parsed.strftime("%H:%M")
        if normalized not in result:
            result.append(normalized)

    if len(result) > 8:
        raise ValueError("Для одного типа сборщика можно выбрать не больше 8 вариантов времени.")

    return sorted(result)


def _collector_cfg(settings, collector):
    collectors = settings.setdefault("collectors", {})
    cfg = collectors.get(collector)
    if not isinstance(cfg, dict):
        cfg = _default_collector_settings()
        collectors[collector] = cfg
    return cfg


def _days_text(days):
    days = sorted(set(days or []))
    if not days:
        return "не выбраны"
    return ", ".join(DAY_LABELS[i] for i in days if 0 <= i <= 6)


def _times_text(times):
    return ", ".join(times or []) if times else "не выбрано"

    lines = [
