            return []
        with DELETE_QUEUE_PATH.open("r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, list) else []
    except Exception as exc:
        print("DELETE QUEUE READ ERROR:", exc, flush=True)
        return []


def _save_delete_queue(queue):
    tmp = DELETE_QUEUE_PATH.with_suffix(".tmp")
    try:
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(queue, f, ensure_ascii=False, indent=2)
        tmp.replace(DELETE_QUEUE_PATH)
    except Exception as exc:
        print("DELETE QUEUE WRITE ERROR:", exc, flush=True)


def _chat_delete_after_minutes(chat_id):
    try:
        _, settings = _get_chat_notify_settings(chat_id)
        value = settings.get("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES)
        value = int(value)
        return max(0, value)
    except Exception:
        return DEFAULT_DELETE_AFTER_MINUTES


def _remove_scheduled_delete(chat_id, message_id):
    with _delete_queue_lock:
        queue = _load_delete_queue()
        filtered = [
            x for x in queue
            if not (
                str(x.get("chat_id")) == str(chat_id)
                and x.get("message_id") == message_id
            )
        ]
        if len(filtered) != len(queue):
            _save_delete_queue(filtered)


def delete_message_later(chat_id, message_id, delay=None):
    """
    Ставит удаление в постоянную очередь.
    delay=None -> использовать настройку конкретного чата.
    delay=0 -> не удалять.
    """
    if delay is None:
        delay = _chat_delete_after_minutes(chat_id) * 60

    try:
        delay = int(delay)
    except Exception:
        delay = DEFAULT_DELETE_AFTER_MINUTES * 60

    if delay <= 0:
        _remove_scheduled_delete(chat_id, message_id)
        return

    item = {
        "chat_id": chat_id,
        "message_id": message_id,
        "delete_at": int(time.time()) + delay,
        "retries": 0,
    }

    with _delete_queue_lock:
        queue = _load_delete_queue()

        # Не плодим дубль для одного сообщения.
        queue = [
            x for x in queue
            if not (
                str(x.get("chat_id")) == str(chat_id)
                and x.get("message_id") == message_id
            )
        ]
        queue.append(item)
        _save_delete_queue(queue)


def _delete_queue_loop():
    while True:
        try:
            now = int(time.time())
            with _delete_queue_lock:
                queue = _load_delete_queue()
                remaining = []

                for item in queue:
                    if int(item.get("delete_at", 0)) > now:
                        remaining.append(item)
                        continue

                    try:
                        telegram_api(
                            "deleteMessage",
                            {
                                "chat_id": item.get("chat_id"),
                                "message_id": item.get("message_id"),
                            },
                        )
                    except Exception as exc:
                        text = str(exc).lower()

                        # Сообщения уже нет — задача фактически выполнена.
                        if "message to delete not found" in text:
                            continue

                        retries = int(item.get("retries", 0)) + 1
                        if retries <= 5:
                            item["retries"] = retries
                            item["delete_at"] = now + 60
                            remaining.append(item)
                        else:
                            print("DELETE GIVE UP:", exc, flush=True)

                _save_delete_queue(remaining)

        except Exception as exc:
            print("DELETE QUEUE LOOP ERROR:", exc, flush=True)

        time.sleep(15)


def _start_delete_queue_worker():
    thread = threading.Thread(
        target=_delete_queue_loop,
        daemon=True,
    )
    thread.start()
