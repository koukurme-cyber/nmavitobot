    message_id = message.get("message_id")
    if chat_id is None or message_id is None:
        return

        minutes = int(data.split(":", 1)[1])
    except Exception:
        return

    if minutes < 0 or minutes > 10080:
        return

    all_settings, settings = _get_chat_notify_settings(chat_id)
    settings["delete_after_minutes"] = minutes
    all_settings[str(chat_id)] = settings
    _save_notify_settings(all_settings)

    # Пересчитываем срок удаления самого экрана настройки уже по новому значению.
    delete_message_later(chat_id, message_id)

    try:
        telegram_api(
            "answerCallbackQuery",
            {"callback_query_id": callback["id"]},
        )
    except Exception:
        pass

    try:
        telegram_api(
            "editMessageText",
            {
                "chat_id": chat_id,
                "message_id": message_id,
                "text": _delete_after_text(settings),
                "parse_mode": "HTML",
                "reply_markup": json.dumps(
                    _delete_after_keyboard(settings),
                    ensure_ascii=False,
                ),
            },
        )
    except Exception as exc:
        if "message is not modified" not in str(exc).lower():
            print("DELETE SETTINGS EDIT ERROR:", exc, flush=True)

    "blue": "nm_blue",
    "tir": "tir",

def _parse_direct_command(command):
    """
    Команды только в прямом формате:
    /status_avmex
    /balance_avmex
    /ads_avmex
    и *_all для всех аккаунтов.
    """
    if command == "/status_all":
        return "full", None
    if command == "/balance_all":
        return "finance", None
    if command == "/ads_all":
        return "ads", None

    for suffix, seller_key in ACCOUNT_COMMAND_SUFFIXES.items():
        if command == f"/status_{suffix}":
            return "full", seller_key
        if command == f"/balance_{suffix}":
            return "finance", seller_key
        if command == f"/ads_{suffix}":
            return "ads", seller_key

    return None, None

def _utf16_len(value):
    return len(value.encode("utf-16-le")) // 2


def _help_command_entities(text):
    commands = [
        "/status_all",
        "/balance_all",
        "/ads_all",
        "/status_avmex",
        "/balance_avmex",
        "/ads_avmex",
        "/delete_after",
        "/commands",
    ]

    entities = []
    search_from = 0
    for command in commands:
        pos = text.find(command, search_from)
        if pos < 0:
            pos = text.find(command)
        if pos < 0:
            continue

        entities.append({
            "type": "bot_command",
            "offset": _utf16_len(text[:pos]),
            "length": _utf16_len(command),
        })
        search_from = pos + len(command)

    return entities

        send_query(chat_id, "full", None)

        send_query(chat_id, "ads", None)
