                save_cached_status(data, fetched_at)

            message = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": _scheduled_status_text(data, fetched_at, collector),
                    "parse_mode": "HTML",
                    "disable_web_page_preview": "true",
                },
            )
            delete_message_later(
                chat_id,
                message["message_id"],
                delay=FIXED_NOTIFICATION_TTL_SECONDS,
            )

            print(
                f"Scheduled notification sent: chat={chat_id}, "
                f"collector={collector}, time={scheduled_time}",
                flush=True,
            )
        except Exception as exc:
            print(
                f"SCHEDULED NOTIFICATION ERROR chat={chat_id}, "
                f"collector={collector}, time={scheduled_time}: {exc}",
                flush=True,
            )


def _notification_scheduler_loop():
    """
    Фиксированное расписание, только Пн–Пт:
      08:00 -> status_all
      08:40 -> balance_all
      10:00..16:00 каждый час -> balance_all

    Пользовательские настройки уведомлений отключены.
    notification_settings.json используется как реестр чатов и для настроек удаления.
    """
    while True:
        try:
            tz_name = CONFIG.get("timezone", "Europe/Moscow")
            now = datetime.now(ZoneInfo(tz_name))

            if now.weekday() < 5:
                all_settings = _load_notify_settings()
                changed = False
                today_str = now.date().isoformat()

                for chat_key, raw_settings in list(all_settings.items()):
                    settings, migrated = _migrate_chat_notify_settings(raw_settings)
                    if migrated:
                        all_settings[chat_key] = settings
                        changed = True

                    fixed_last_sent = settings.setdefault(
                        "fixed_schedule_last_sent",
                        {},
                    )

                    for time_str, collector in FIXED_WEEKDAY_SCHEDULE.items():
                        hh, mm = [int(x) for x in time_str.split(":", 1)]
                        scheduled_dt = now.replace(
                            hour=hh,
                            minute=mm,
                            second=0,
                            microsecond=0,
                        )
                        seconds_late = (now - scheduled_dt).total_seconds()

                        # После короткого рестарта допускаем запоздание до 3 минут.
                        if not (0 <= seconds_late < 180):
                            continue

                        if fixed_last_sent.get(time_str) == today_str:
                            continue

                        fixed_last_sent[time_str] = today_str
                        changed = True

                        try:
                            chat_id = int(chat_key)
                        except ValueError:
                            chat_id = chat_key

                        threading.Thread(
                            target=_send_scheduled_status,
                            args=(chat_id, collector, time_str),
                            daemon=True,
                        ).start()

                if changed:
                    _save_notify_settings(all_settings)

        except Exception as exc:
            print("FIXED NOTIFICATION SCHEDULER ERROR:", exc, flush=True)

        time.sleep(15)


def _start_notification_scheduler():
    thread = threading.Thread(
        target=_notification_scheduler_loop,
        daemon=True,
    )
    thread.start()


def _answer_callback(callback, text=None, alert=False):
    payload = {"callback_query_id": callback["id"]}
    if text:
        payload["text"] = text
        payload["show_alert"] = "true" if alert else "false"
    try:
        telegram_api("answerCallbackQuery", payload)
    except Exception:
        pass

        return
