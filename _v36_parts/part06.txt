                        continue

                    if last_sent.get(time_str) == today_str:
                        continue

                    # Отмечаем до отправки, чтобы параллельный цикл не дал дубль.
                    last_sent[time_str] = today_str
                    changed = True
                    try:
                        chat_id = int(chat_key)
                    except ValueError:
                        chat_id = chat_key

                    threading.Thread(
                        target=_send_scheduled_status,
                        args=(chat_id, time_str),
                        daemon=True,
                    ).start()

            if changed:
                _save_notify_settings(all_settings)

        except Exception as exc:
            print("NOTIFICATION SCHEDULER ERROR:", exc, flush=True)

        time.sleep(15)


def _start_notification_scheduler():
    thread = threading.Thread(
        target=_notification_scheduler_loop,
        daemon=True,
    )
    thread.start()


def _handle_notify_callback(callback):
    data = callback.get("data") or ""
    message = callback.get("message") or {}
    chat = message.get("chat") or {}
    chat_id = chat.get("id")
    message_id = message.get("message_id")
    if chat_id is None or message_id is None:
        return

    all_settings, settings = _get_chat_notify_settings(chat_id)

    if data.startswith("notify:day:"):
        try:
            day = int(data.rsplit(":", 1)[1])
        except ValueError:
            return
        days = set(settings.get("days") or [])
        if day in days:
            days.remove(day)
        else:
            days.add(day)
        settings["days"] = sorted(days)

    elif data.startswith("notify:time:"):
        raw = data.rsplit(":", 1)[1]
        if len(raw) != 4 or not raw.isdigit():
            return
        value = f"{raw[:2]}:{raw[2:]}"
        try:
            value = _normalize_times([value])[0]
        except Exception:
            return

        times = list(settings.get("times") or [])
        if value in times:
            times.remove(value)
        else:
            if len(times) >= 8:
                telegram_api(
                    "answerCallbackQuery",
                    {
                        "callback_query_id": callback["id"],
                        "text": "Можно выбрать не больше 8 вариантов времени.",
                        "show_alert": "true",
                    },
                )
                return
            times.append(value)
        settings["times"] = sorted(times)

    elif data == "notify:clear_times":
        settings["times"] = []

    elif data == "notify:noop":
        try:
            telegram_api(
                "answerCallbackQuery",
                {"callback_query_id": callback["id"]},
            )
        except Exception:
            pass
        return

    elif data == "notify:weekdays":
        settings["days"] = [0, 1, 2, 3, 4]
    elif data == "notify:all_days":
        settings["days"] = [0, 1, 2, 3, 4, 5, 6]
    elif data == "notify:no_days":
        settings["days"] = []
    elif data == "notify:toggle":
        if not settings.get("enabled"):
            if not settings.get("days"):
                telegram_api(
                    "answerCallbackQuery",
                    {
                        "callback_query_id": callback["id"],
                        "text": "Сначала выберите хотя бы один день.",
                        "show_alert": "true",
                    },
                )
                return
            if not settings.get("times"):
                telegram_api(
                    "answerCallbackQuery",
                    {
                        "callback_query_id": callback["id"],
                        "text": "Сначала задайте хотя бы одно время через /notify_times.",
                        "show_alert": "true",
                    },
                )
                return
        settings["enabled"] = not bool(settings.get("enabled"))

    all_settings[str(chat_id)] = settings
    _save_notify_settings(all_settings)

    try:
        telegram_api(
            "answerCallbackQuery",
            {"callback_query_id": callback["id"]},
        )
    except Exception:
        pass

    _edit_notify_settings(chat_id, message_id, settings)



DELETE_PRESETS = [5, 10, 15, 20, 30, 45, 60, 90, 120, 180]


def _delete_after_text(settings):
    minutes = int(settings.get("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES))
    if minutes <= 0:
        current = "не удалять автоматически"
    else:
        current = f"{minutes} мин."

    return (
        "<b>АВИТО · ВРЕМЯ ПОКАЗА СООБЩЕНИЙ</b>\n\n"
        f"Сейчас: {current}\n\n"
        "Просто нажмите нужное время ниже.\n"
        "Настройка применяется к новым ответам бота и новым командам пользователя.\n\n"
        "Для нестандартного значения остаётся команда:\n"
        "/delete_after 75"
    )


def _delete_after_keyboard(settings):
    current = int(settings.get("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES))
    buttons = []
    for value in DELETE_PRESETS:
        text = f"{'✅ ' if current == value else ''}{value} мин"
        buttons.append({
            "text": text,
            "callback_data": f"delete:{value}",
        })

    return {
        "inline_keyboard": [
            buttons[:3],
            buttons[3:6],
            buttons[6:9],
            buttons[9:],
            [{
                "text": ("✅ " if current == 0 else "") + "Не удалять",
                "callback_data": "delete:0",
            }],
        ]
    }


def _send_delete_after_settings(chat_id):
    all_settings, settings = _get_chat_notify_settings(chat_id)
    message = telegram_api(

        {
            "chat_id": chat_id,

            "parse_mode": "HTML",
            "reply_markup": json.dumps(
                _delete_after_keyboard(settings),
                ensure_ascii=False,
            ),
        },
    )
    delete_message_later(chat_id, message["message_id"])
    return message
