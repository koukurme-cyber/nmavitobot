
                    threading.Thread(
                        target=_send_scheduled_status,
                        args=(chat_id, time_str),
                        daemon=True,
                    ).start()

            if changed:
                _save_notify_settings(all_settings)

        except Exception as exc:
            print("NOTIFICATION SCHEDULER ERROR:", exc, flush=True)

        time.sleep(15)

def _handle_notify_callback(callback):
    data = callback.get("data") or ""
    message = callback.get("message") or {}
    chat = message.get("chat") or {}
    chat_id = chat.get("id")
    message_id = message.get("message_id")
    if chat_id is None or message_id is None:
        return

    all_settings, settings = _get_chat_notify_settings(chat_id)

    if data.startswith("notify:day:"):
        try:
            day = int(data.rsplit(":", 1)[1])
        except ValueError:
            return
        days = set(settings.get("days") or [])
        if day in days:
            days.remove(day)
        else:
            days.add(day)
        settings["days"] = sorted(days)

    elif data == "notify:weekdays":
        settings["days"] = [0, 1, 2, 3, 4]
    elif data == "notify:all_days":
        settings["days"] = [0, 1, 2, 3, 4, 5, 6]
    elif data == "notify:no_days":
        settings["days"] = []
    elif data == "notify:toggle":
        if not settings.get("enabled"):
            if not settings.get("days"):
                telegram_api(
                    "answerCallbackQuery",
                    {
                        "callback_query_id": callback["id"],
                        "text": "Сначала выберите хотя бы один день.",
                        "show_alert": "true",
                    },
                )
                return
            if not settings.get("times"):
                telegram_api(
                    "answerCallbackQuery",
                    {
                        "callback_query_id": callback["id"],
                        "text": "Сначала задайте хотя бы одно время через /notify_times.",
                        "show_alert": "true",
                    },
                )
                return
        settings["enabled"] = not bool(settings.get("enabled"))

    all_settings[str(chat_id)] = settings
    _save_notify_settings(all_settings)

    try:
        telegram_api(
            "answerCallbackQuery",
            {"callback_query_id": callback["id"]},
        )
    except Exception:
        pass

    _edit_notify_settings(chat_id, message_id, settings)



DELETE_PRESETS = [5, 10, 20, 30, 60, 120]


def _delete_after_text(settings):
    minutes = int(settings.get("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES))
    if minutes <= 0:
        current = "не удалять автоматически"
    else:
        current = f"{minutes} мин."

    return (
        "<b>АВИТО · АВТОУДАЛЕНИЕ</b>\n\n"
        f"Сейчас: {current}\n\n"
        "Выберите вариант кнопкой ниже или задайте своё значение:\n"
        "<code>/delete_after 45</code>\n\n"
        "<code>/delete_after 0</code> — не удалять автоматически."
    )


def _delete_after_keyboard(settings):
    current = int(settings.get("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES))
    buttons = []
    for value in DELETE_PRESETS:
        text = f"{'✅ ' if current == value else ''}{value} мин"
        buttons.append({
            "text": text,
            "callback_data": f"delete:{value}",
        })

    return {
        "inline_keyboard": [
            buttons[:3],
            buttons[3:],
            [{
                "text": ("✅ " if current == 0 else "") + "Не удалять",
                "callback_data": "delete:0",
            }],
        ]
    }

def _handle_delete_callback(callback):
    data = callback.get("data") or ""
    message = callback.get("message") or {}
    chat_id = (message.get("chat") or {}).get("id")
    message_id = message.get("message_id")
    if chat_id is None or message_id is None:
        return

    try:
        minutes = int(data.split(":", 1)[1])
    except Exception:
        return

    if minutes < 0 or minutes > 10080:
        return

    all_settings, settings = _get_chat_notify_settings(chat_id)
    settings["delete_after_minutes"] = minutes
    all_settings[str(chat_id)] = settings
    _save_notify_settings(all_settings)

    try:
        telegram_api(
            "answerCallbackQuery",
            {"callback_query_id": callback["id"]},

    except Exception:
        pass

    try:
        telegram_api(
            "editMessageText",

                "chat_id": chat_id,
                "message_id": message_id,
                "text": _delete_after_text(settings),
                "parse_mode": "HTML",

                    _delete_after_keyboard(settings),
                    ensure_ascii=False,
                ),
            },
        )
    except Exception as exc:
        if "message is not modified" not in str(exc).lower():
            print("DELETE SETTINGS EDIT ERROR:", exc, flush=True)


ACCOUNT_COMMAND_SUFFIXES = {
    "aggregaty": "aggregaty",
    "avmex": "avmex",
    "orange": "nm_orange",
    "blue": "nm_blue",
    "tir": "tir",
}


def _parse_direct_command(command):
    """
    Команды только в прямом формате:
    /status_avmex
    /balance_avmex
    /ads_avmex
    и *_all для всех аккаунтов.
    """
    if command == "/status_all":
        return "full", None
    if command == "/balance_all":
        return "finance", None
    if command == "/ads_all":
        return "ads", None

    for suffix, seller_key in ACCOUNT_COMMAND_SUFFIXES.items():
        if command == f"/status_{suffix}":
            return "full", seller_key
        if command == f"/balance_{suffix}":
            return "finance", seller_key
        if command == f"/ads_{suffix}":
            return "ads", seller_key

    return None, None
