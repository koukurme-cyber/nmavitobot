    while True:
        try:
            now = int(time.time())
            with _delete_queue_lock:
                queue = _load_delete_queue()
                remaining = []

                for item in queue:
                    if int(item.get("delete_at", 0)) > now:
                        remaining.append(item)
                        continue

                    try:
                        telegram_api(
                            "deleteMessage",
                            {
                                "chat_id": item.get("chat_id"),
                                "message_id": item.get("message_id"),
                            },
                        )
                    except Exception as exc:
                        text = str(exc).lower()

                        # Сообщения уже нет — задача фактически выполнена.
                        if "message to delete not found" in text:
                            continue

                        retries = int(item.get("retries", 0)) + 1
                        if retries <= 5:
                            item["retries"] = retries
                            item["delete_at"] = now + 60
                            remaining.append(item)
                        else:
                            print("DELETE GIVE UP:", exc, flush=True)

                _save_delete_queue(remaining)

        except Exception as exc:
            print("DELETE QUEUE LOOP ERROR:", exc, flush=True)

        time.sleep(15)


def _start_delete_queue_worker():
    thread = threading.Thread(
        target=_delete_queue_loop,
        daemon=True,
    )
    thread.start()


def _send_final_query_result(chat_id, mode="full", seller_key=None):
    # Пользователь ничего не видит, пока все данные не собраны.
    with _refresh_lock:
        try:
            tz_name = CONFIG.get("timezone", "Europe/Moscow")
            query_config = _config_for_seller(seller_key)

            if mode == "finance":
                data = get_financial_status(query_config)
                renderer = render_account_balance if seller_key else render_balances
            elif mode == "ads":
                data = get_items_status(query_config)
                renderer = render_account_ads if seller_key else render_ads
            else:
                data = get_status(query_config)
                renderer = render_account if seller_key else render_status

            if mode != "ads":
                data = _apply_payment_dates(data)

            fetched_at = datetime.now(ZoneInfo(tz_name)).strftime("%d.%m.%Y %H:%M")

            if mode == "full" and not seller_key:
                save_cached_status(data, fetched_at)

            message = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": renderer(data, fetched_at),
                    "parse_mode": "HTML",
                    "disable_web_page_preview": "true",
                },
            )
            delete_message_later(chat_id, message["message_id"])

            print(
                f"Telegram sent final: mode={mode}, seller={seller_key or 'all'}",
                flush=True,
            )

        except Exception as exc:
            print("BACKGROUND REFRESH ERROR:", exc, flush=True)
            try:
                message = telegram_api(
                    "sendMessage",
                    {
                        "chat_id": chat_id,
                        "text": "<b>АВИТО</b>\n\nНе удалось получить актуальные данные.",
                        "parse_mode": "HTML",
                    },
                )
                delete_message_later(chat_id, message["message_id"])
            except Exception as send_exc:
                print("ERROR SEND ERROR:", send_exc, flush=True)


def start_background_query(chat_id, mode="full", seller_key=None):
    thread = threading.Thread(
        target=_send_final_query_result,
        args=(chat_id, mode, seller_key),
        daemon=True,
    )
    thread.start()

def _load_notify_settings():
    try:
        if not NOTIFY_SETTINGS_PATH.exists():
            return {}
        with NOTIFY_SETTINGS_PATH.open("r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, dict) else {}
    except Exception as exc:
        print("NOTIFY SETTINGS READ ERROR:", exc, flush=True)
        return {}


def _save_notify_settings(settings):
    tmp = NOTIFY_SETTINGS_PATH.with_suffix(".tmp")
    try:
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(settings, f, ensure_ascii=False, indent=2)
        tmp.replace(NOTIFY_SETTINGS_PATH)
    except Exception as exc:
        print("NOTIFY SETTINGS WRITE ERROR:", exc, flush=True)


def _default_notify_settings():
    return {
        "enabled": False,
        "days": [0, 1, 2, 3, 4, 5, 6],
        "times": [],
        "last_sent": {},
        "delete_after_minutes": DEFAULT_DELETE_AFTER_MINUTES,
    }


def _get_chat_notify_settings(chat_id):
    all_settings = _load_notify_settings()
    key = str(chat_id)
    current = all_settings.get(key)
    if not isinstance(current, dict):
        current = _default_notify_settings()
        all_settings[key] = current
        _save_notify_settings(all_settings)

    current.setdefault("enabled", False)
    current.setdefault("days", [0, 1, 2, 3, 4, 5, 6])
    current.setdefault("times", [])
    current.setdefault("last_sent", {})
    current.setdefault("delete_after_minutes", DEFAULT_DELETE_AFTER_MINUTES)
    return all_settings, current


def _normalize_times(values):
    result = []
    for value in values:
        value = str(value).strip()
        try:
            parsed = datetime.strptime(value, "%H:%M")
        except ValueError:
            raise ValueError(f"Неверное время: {value}. Нужен формат ЧЧ:ММ.")
        normalized = parsed.strftime("%H:%M")
        if normalized not in result:
            result.append(normalized)

    if len(result) > 8:
        raise ValueError("Можно выбрать не больше 8 вариантов времени.")

    return sorted(result)
