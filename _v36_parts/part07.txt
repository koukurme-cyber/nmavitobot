def handle_update(update):
    message = update.get("message")
    if not message:
        return

    text = (message.get("text") or "").strip()
    chat_id = message["chat"]["id"]
    if not text.startswith("/"):
        return

    command = text.split(maxsplit=1)[0].split("@", 1)[0].casefold()

    # Команда пользователя исчезает по той же настройке, что и ответы бота.
    delete_message_later(chat_id, message["message_id"])

    if command == "/notify":
        _send_notify_settings(chat_id)
        return

    if command == "/notify_times":
        parts = text.split(maxsplit=1)
        raw_args = parts[1] if len(parts) > 1 else ""

        if not raw_args.strip():
            _, settings = _get_chat_notify_settings(chat_id)
            current = ", ".join(settings.get("times") or []) or "не выбрано"
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": (
                        "<b>АВИТО · ВРЕМЯ УВЕДОМЛЕНИЙ</b>\n\n"
                        f"Сейчас: {html.escape(current)}\n\n"
                        "Задайте до 8 времён одной командой, например:\n"
                        "<code>/notify_times 09:00 13:30 18:00</code>"
                    ),
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        try:
            times = _normalize_times(raw_args.replace(",", " ").split())
        except ValueError as exc:
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": html.escape(str(exc)),
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        all_settings, settings = _get_chat_notify_settings(chat_id)
        settings["times"] = times
        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)
        _send_notify_settings(chat_id)
        return

    if command == "/delete_after":
        parts = text.split(maxsplit=1)

        if len(parts) == 1:
            _send_delete_after_settings(chat_id)
            return

        raw = parts[1].strip()
        try:
            minutes = int(raw)
        except ValueError:
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": "Укажите число минут, например <code>/delete_after 45</code>.",
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        if minutes < 0 or minutes > 10080:
            reply = telegram_api(
                "sendMessage",
                {
                    "chat_id": chat_id,
                    "text": "Допустимо от 0 до 10080 минут.",
                    "parse_mode": "HTML",
                },
            )
            delete_message_later(chat_id, reply["message_id"])
            return

        all_settings, settings = _get_chat_notify_settings(chat_id)
        settings["delete_after_minutes"] = minutes
        all_settings[str(chat_id)] = settings
        _save_notify_settings(all_settings)
        _send_delete_after_settings(chat_id)
        return

    if command == "/start":
        help_message = telegram_api(
            "sendMessage",
            {
                "chat_id": chat_id,
                "text": _commands_help(),
                "parse_mode": "HTML",
            },
        )
        delete_message_later(chat_id, help_message["message_id"])
        return

    mode, seller_key = _parse_direct_command(command)
    if mode:
        send_query(chat_id, mode, seller_key)
        return

    if command in {"/help", "/commands"}:
        help_message = telegram_api(
            "sendMessage",
            {
                "chat_id": chat_id,
                "text": _commands_help(),
                "parse_mode": "HTML",
            },
        )
        delete_message_later(chat_id, help_message["message_id"])
